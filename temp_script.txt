    async function fetchListings() {
      const res = await fetch(`${supabaseUrl}/rest/v1/listings?select=*&order=created_at.desc`, {
        headers: {
          'apikey': supabaseKey,
          'Authorization': `Bearer ${supabaseKey}`
        }
      });
      if (!res.ok) throw new Error('Failed to fetch listings');
      return res.json();

    async function updateFollowed(listingId, followed) {
      const res = await fetch(`${supabaseUrl}/rest/v1/listings?id=eq.${listingId}`, {
        method: 'PATCH',
        headers: {
          'apikey': supabaseKey,
          'Authorization': `Bearer ${supabaseKey}`,
          'Content-Type': 'application/json',
          'Prefer': 'return=minimal'
        },
        body: JSON.stringify({ followed })

      return res.ok;
    }

              æˆ‘çš„å…³æ³¨ ({followedIds.size})
            </button>
            <button className={`nav-tab wishlist ${activeTab === 'wishlist' ? 'active' : ''}`} onClick={() => { setActiveTab('wishlist'); setFilter('wishlist'); }}>
              å¿ƒæ„¿å• ({wishlist.size})
            </button>
